# Left Side Nav Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a fixed-width left sidebar with a date-derived folder tree (Library → Year → Month) and a flat draggable albums list, wired into the existing `Library.tsx` page.

**Architecture:** Date tree is derived client-side from the `photos` array already in Zustand — no new Rust commands for that. Albums get a new Rust module (`commands/albums.rs`) and a `useAlbums` hook. Drag-and-drop uses `@dnd-kit` in a single `<DndContext>` covering both the sidebar and the grid. A unified `activeSource` discriminated union replaces `activeAlbumId`, giving mutually-exclusive selection across the tree and albums list.

**Tech Stack:** React 19, TypeScript strict, Zustand, Tailwind, Radix UI, `@dnd-kit/core` + `@dnd-kit/sortable`, Tauri 2.x, sqlx + SQLite, vitest.

**Reference spec:** `docs/superpowers/specs/2026-04-19-side-nav-design.md`

```ts
import { create } from "zustand";
import type { ActiveSource, Album, Photo } from "../types";

const SPLIT_STORAGE_KEY = "archiva.sideNavSplit";

function loadSplit(): number {
  if (typeof window === "undefined") return 50;
  const raw = window.localStorage.getItem(SPLIT_STORAGE_KEY);
  if (!raw) return 50;
  const n = Number(raw);
  if (!Number.isFinite(n)) return 50;
  return Math.min(80, Math.max(20, n));
}

function saveSplit(n: number) {
  if (typeof window === "undefined") return;
  window.localStorage.setItem(SPLIT_STORAGE_KEY, String(n));
}

interface LibraryState {
  photos: Photo[];
  albums: Album[];
  selectedIds: Set<string>;
  activeSource: ActiveSource;
  activePhotoId: string | null;
  expandedYears: Set<number>;
  foldersHeightPct: number;
  albumPhotoIds: Map<string, Set<string>>;

  setPhotos: (photos: Photo[]) => void;
  setAlbums: (albums: Album[]) => void;
  selectPhoto: (id: string) => void;
  toggleSelect: (id: string) => void;
  clearSelection: () => void;
  setActiveSource: (s: ActiveSource) => void;
  toggleYear: (year: number) => void;
  setFoldersHeightPct: (n: number) => void;
  setAlbumPhotoIds: (albumId: string, ids: Set<string>) => void;
}

export const useLibraryStore = create<LibraryState>((set) => ({
  photos: [],
  albums: [],
  selectedIds: new Set<string>(),
  activeSource: { kind: "library" },
  activePhotoId: null,
  expandedYears: new Set<number>(),
  foldersHeightPct: loadSplit(),
  albumPhotoIds: new Map<string, Set<string>>(),

  setPhotos: (photos) => set({ photos }),
  setAlbums: (albums) => set({ albums }),
  selectPhoto: (id) =>
    set({ selectedIds: new Set([id]), activePhotoId: id }),
  toggleSelect: (id) =>
    set((state) => {
      const next = new Set(state.selectedIds);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return { selectedIds: next };
    }),
  clearSelection: () => set({ selectedIds: new Set(), activePhotoId: null }),
  setActiveSource: (s) => set({ activeSource: s }),
  toggleYear: (year) =>
    set((state) => {
      const next = new Set(state.expandedYears);
      if (next.has(year)) next.delete(year);
      else next.add(year);
      return { expandedYears: next };
    }),
  setFoldersHeightPct: (n) => {
    const clamped = Math.min(80, Math.max(20, n));
    saveSplit(clamped);
    set({ foldersHeightPct: clamped });
  },
  setAlbumPhotoIds: (albumId, ids) =>
    set((state) => {
      const next = new Map(state.albumPhotoIds);
      next.set(albumId, ids);
      return { albumPhotoIds: next };
    }),
}));
```

```ts
import { useMemo } from "react";
import type { DateTree, MonthNode, Photo, YearNode } from "../types";

export function buildDateTree(photos: Photo[]): DateTree {
  const byYear = new Map<number, Map<number, number>>();
  let noDateCount = 0;

  for (const photo of photos) {
    const iso = photo.created_at ?? photo.modified_at;
    if (!iso) {
      noDateCount += 1;
      continue;
    }
    const d = new Date(iso);
    const time = d.getTime();
    if (!Number.isFinite(time)) {
      noDateCount += 1;
      continue;
    }
    const year = d.getUTCFullYear();
    const month = d.getUTCMonth() + 1;
    const months = byYear.get(year) ?? new Map<number, number>();
    months.set(month, (months.get(month) ?? 0) + 1);
    byYear.set(year, months);
  }

  const years: YearNode[] = Array.from(byYear.entries())
    .map(([year, months]) => {
      const monthNodes: MonthNode[] = Array.from(months.entries())
        .map(([month, count]) => ({ month, count }))
        .sort((a, b) => b.month - a.month);
      const count = monthNodes.reduce((sum, m) => sum + m.count, 0);
      return { year, count, months: monthNodes };
    })
    .sort((a, b) => b.year - a.year);

  return { total: photos.length, years, noDateCount };
}

export function useDateTree(photos: Photo[]): DateTree {
  return useMemo(() => buildDateTree(photos), [photos]);
}
```

```ts
import { useMemo } from "react";
import type { ActiveSource, Photo } from "../types";
import { useLibraryStore } from "../store/useLibraryStore";

export function filterPhotos(
  photos: Photo[],
  src: ActiveSource,
  albumPhotoIds: Map<string, Set<string>>,
): Photo[] {
  switch (src.kind) {
    case "library":
      return photos;
    case "no-date":
      return photos.filter((p) => !p.created_at && !p.modified_at);
    case "date":
      return photos.filter((p) => matchesDate(p, src.year, src.month));
    case "album": {
      const ids = albumPhotoIds.get(src.id);
      if (!ids) return [];
      return photos.filter((p) => ids.has(p.id));
    }
  }
}

function matchesDate(p: Photo, year: number, month?: number): boolean {
  const iso = p.created_at ?? p.modified_at;
  if (!iso) return false;
  const d = new Date(iso);
  if (!Number.isFinite(d.getTime())) return false;
  if (d.getUTCFullYear() !== year) return false;
  if (month !== undefined && d.getUTCMonth() + 1 !== month) return false;
  return true;
}

export function useVisiblePhotos(): Photo[] {
  const photos = useLibraryStore((s) => s.photos);
  const activeSource = useLibraryStore((s) => s.activeSource);
  const albumPhotoIds = useLibraryStore((s) => s.albumPhotoIds);
  return useMemo(
    () => filterPhotos(photos, activeSource, albumPhotoIds),
    [photos, activeSource, albumPhotoIds],
  );
}
```

```rust
            .bind(album_id)

            .await?;

    let now = Utc::now().to_rfc3339();
    let mut inserted = 0usize;
```

**Files:**

- [ ] **Step 1: Implement**

Create `src/hooks/useAlbums.ts`:

```ts
import { useCallback, useEffect } from "react";
import {
  addPhotosToAlbum as apiAddPhotos,
  listAlbumPhotos as apiListAlbumPhotos,
  listAlbums as apiListAlbums,
  reorderAlbums as apiReorderAlbums,
} from "../lib/tauri";
import { useLibraryStore } from "../store/useLibraryStore";
import type { Album } from "../types";

export function useAlbums() {
  const albums = useLibraryStore((s) => s.albums);
  const setAlbums = useLibraryStore((s) => s.setAlbums);
  const activeSource = useLibraryStore((s) => s.activeSource);
  const albumPhotoIds = useLibraryStore((s) => s.albumPhotoIds);
  const setAlbumPhotoIds = useLibraryStore((s) => s.setAlbumPhotoIds);

  const refresh = useCallback(async () => {
    const rows = await apiListAlbums();
    setAlbums(rows);
  }, [setAlbums]);

  useEffect(() => {
    void refresh();
  }, [refresh]);

  // Hydrate album photo ids when an album is activated (cache first).
  useEffect(() => {
    if (activeSource.kind !== "album") return;
    if (albumPhotoIds.has(activeSource.id)) return;
    const id = activeSource.id;
    void apiListAlbumPhotos(id).then((ids) => {
      setAlbumPhotoIds(id, new Set(ids));
    });
  }, [activeSource, albumPhotoIds, setAlbumPhotoIds]);

  const reorder = useCallback(
    async (orderedIds: string[]) => {
      const prev = albums;
      // Optimistic reorder
      const byId = new Map(prev.map((a) => [a.id, a]));
      const next: Album[] = orderedIds
        .map((id) => byId.get(id))
        .filter((a): a is Album => !!a)
        .map((a, i) => ({ ...a, sort_order: i }));
      setAlbums(next);
      try {
        await apiReorderAlbums(orderedIds);
      } catch (e) {
        setAlbums(prev);
        throw e;
      }
    },
    [albums, setAlbums],
  );

  const addPhotos = useCallback(
    async (albumId: string, photoIds: string[]) => {
      const count = await apiAddPhotos(albumId, photoIds);
      // Invalidate cache so the next activation re-fetches.
      if (albumPhotoIds.has(albumId)) {
        const ids = await apiListAlbumPhotos(albumId);
        setAlbumPhotoIds(albumId, new Set(ids));
      }
      return count;
    },
    [albumPhotoIds, setAlbumPhotoIds],
  );

  return { albums, refresh, reorder, addPhotos };
}
```

```tsx
        setDropRef(node);

      className={clsx(

          isActive ? "bg-accent text-white" : "hover:bg-panel-hover",
```

```tsx
import { useEffect, useRef, type RefObject } from "react";
import { useLibraryStore } from "../../store/useLibraryStore";

interface Props {
  containerRef: RefObject<HTMLElement | null>;
}

export function SectionResizer({ containerRef }: Props) {
  const setFoldersHeightPct = useLibraryStore((s) => s.setFoldersHeightPct);
  const draggingRef = useRef(false);

  useEffect(() => {
    function onMove(e: PointerEvent) {
      if (!draggingRef.current) return;
      const el = containerRef.current;
      if (!el) return;
      const rect = el.getBoundingClientRect();
      if (rect.height <= 0) return;
      const pct = ((e.clientY - rect.top) / rect.height) * 100;
      setFoldersHeightPct(pct);
    }
    function onUp() {
      draggingRef.current = false;
    }
    window.addEventListener("pointermove", onMove);
    window.addEventListener("pointerup", onUp);
    return () => {
      window.removeEventListener("pointermove", onMove);
      window.removeEventListener("pointerup", onUp);
    };
  }, [containerRef, setFoldersHeightPct]);

  return (
    <div
      role="separator"
      aria-orientation="horizontal"
      onPointerDown={(e) => {
        draggingRef.current = true;
        e.currentTarget.setPointerCapture(e.pointerId);
      }}
      className="relative flex h-2 cursor-row-resize items-center justify-center border-y border-border bg-panel select-none"
    >
      <span className="text-[8px] leading-none tracking-[2px] text-muted">⋮⋮</span>
    </div>
  );
}
```

```tsx
import { useMemo } from "react";
import { DndContext, type DragEndEvent, PointerSensor, useSensor, useSensors } from "@dnd-kit/core";
import { arrayMove } from "@dnd-kit/sortable";
import { FolderPicker } from "../components/FolderPicker";
import { MetadataPanel } from "../components/MetadataPanel";
import { PhotoGrid } from "../components/PhotoGrid";
import { SideNav } from "../components/sidenav/SideNav";
import { useAlbums } from "../hooks/useAlbums";
import { usePhotos } from "../hooks/usePhotos";
import { useVisiblePhotos } from "../hooks/useVisiblePhotos";
import { useLibraryStore } from "../store/useLibraryStore";

export function Library() {
  const { loading, error, scan, refresh } = usePhotos();
  const photos = useLibraryStore((s) => s.photos);
  const activePhotoId = useLibraryStore((s) => s.activePhotoId);
  const { albums, reorder, addPhotos } = useAlbums();

  const visible = useVisiblePhotos();
  const activePhoto = useMemo(
    () => photos.find((p) => p.id === activePhotoId) ?? null,
    [photos, activePhotoId],
  );

  const sensors = useSensors(useSensor(PointerSensor, { activationConstraint: { distance: 4 } }));

  async function handleDragEnd(e: DragEndEvent) {
    const { active, over } = e;
    if (!over) return;
    const activeKind = active.data.current?.kind;
    const overData = over.data.current;

    if (activeKind === "album" && overData?.kind !== "album-drop") {
      // Sortable reorder: `over.id` is the album id being displaced.
      if (active.id === over.id) return;
      const oldIndex = albums.findIndex((a) => a.id === active.id);
      const newIndex = albums.findIndex((a) => a.id === over.id);
      if (oldIndex < 0 || newIndex < 0) return;
      const nextIds = arrayMove(albums, oldIndex, newIndex).map((a) => a.id);
      await reorder(nextIds).catch((err) => console.error("reorder failed", err));
      return;
    }

    if (activeKind === "photo" && overData?.kind === "album-drop") {
      const selectedIds = (active.data.current?.selectedIds as string[] | undefined) ?? [];
      if (selectedIds.length === 0) return;
      await addPhotos(overData.albumId as string, selectedIds).catch((err) =>
        console.error("add photos failed", err),
      );
    }
  }

  return (
    <DndContext sensors={sensors} onDragEnd={handleDragEnd}>
      <div className="flex h-screen w-screen flex-col bg-bg text-white">
        <header className="flex items-center justify-between gap-4 border-b border-border px-4 py-2">
          <div className="flex items-center gap-3">
            <h1 className="text-sm font-semibold">Archiva</h1>
            <span className="text-xs text-muted">{visible.length} photos</span>
            {loading && <span className="text-xs text-muted">Loading…</span>}
          </div>
          <div className="flex items-center gap-2">
            <button
              type="button"
              onClick={() => void refresh()}
              className="rounded-md border border-border bg-panel px-3 py-1.5 text-sm text-white hover:border-muted"
            >
              Refresh
            </button>
            <FolderPicker onScan={scan} disabled={loading} />
          </div>
        </header>

        {error && (
          <div className="border-b border-border bg-panel px-4 py-2 text-xs text-red-400">
            {error}
          </div>
        )}

        <div className="flex min-h-0 flex-1">
          <SideNav />
          <main className="min-w-0 flex-1">
            {visible.length === 0 && !loading ? (
              <div className="flex h-full items-center justify-center text-sm text-muted">
                No photos in this view.
              </div>
            ) : (
              <PhotoGrid photos={visible} />
            )}
          </main>
          <aside className="w-72 shrink-0 border-l border-border bg-panel">
            <MetadataPanel photo={activePhoto} />
          </aside>
        </div>
      </div>
    </DndContext>
  );
}
```
